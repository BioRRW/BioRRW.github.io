const daysOfWeek = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"]; const now = new Date(); const currentDayName = daysOfWeek[now.getDay()]; const currentMonthNum = now.getMonth() + 1; const currentDateNum = now.getDate(); let globalPayload = {}; // Enhanced frontend content filter to lock out festivals and non-food events const BLACKLIST = [ "our story", "contact us", "hours", "brewery", "book an event", "event pricing", "harvest host", "what's on tap", "about us", "taproom hours", "back taproom", "back book an event", "denim days", "slush party", "live music", "sundays at odell" ]; fetch('/assets/data/food-trucks.json?v=' + now.getTime()) .then(res => res.json()) .then(payload => { globalPayload = payload; document.getElementById('update-time').innerText = `Scraper Engine Sync: ${payload.last_updated}`; renderDashboard('all'); }) .catch(err => { document.getElementById('update-time').innerText = "System engine synchronization offline."; console.error(err); }); function getNormalizedText(item) { if (!item) return ""; if (item.day_listing) return item.day_listing; if (item.listing) return item.listing; if (item.truck && item.schedule) return `${item.truck} — ${item.schedule}`; return item.truck || ""; } function isToday(text) { const textLower = text.toLowerCase(); const dateStr = `${currentMonthNum}/${currentDateNum}`; const wordStr = `july ${currentDateNum}`; const wordShortStr = `jul ${currentDateNum}`; if (textLower.includes('/') && !textLower.includes(dateStr)) return false; if ((textLower.includes('july') || textLower.includes('jul')) && (!textLower.includes(wordStr) && !textLower.includes(wordShortStr))) return false; return textLower.includes(currentDayName.toLowerCase()) || textLower.includes(currentDayName.substring(0,3).toLowerCase()); } function parseBreweryData(arrayData) { if (!arrayData || arrayData.length === 0) return []; return arrayData.map(item => getNormalizedText(item)) .filter(text => text.length > 12 && !BLACKLIST.some(j => text.toLowerCase().includes(j))); } function renderDashboard(mode) { const mappings = { "Stodgy Brewing": "listings-stodgy", "Maxline Brewing": "listings-maxline", "Zwei Brewing": "listings-zwei", "Mythmaker Brewing": "listings-mythmaker", "Odell Brewing": "listings-odell", "New Belgium": "listings-newbelgium", "Purpose Brewing": "listings-purpose" }; Object.values(mappings).forEach(id => { document.getElementById(id).innerHTML = ""; document.getElementById(id).closest('.brewery-card').style.display = "flex"; }); for (const [name, containerId] of Object.entries(mappings)) { const targetElement = document.getElementById(containerId); const rawItems = globalPayload.breweries ? globalPayload.breweries[name] : []; let lines = parseBreweryData(rawItems); if (mode === 'today') { lines = lines.filter(line => isToday(line)); } if (lines.length === 0) { if (mode === 'today') { document.getElementById(containerId).closest('.brewery-card').style.display = "none"; } else { targetElement.innerHTML = `
Schedule rotating on platform.
`; } } else { lines.forEach(text => { const todayFlag = isToday(text); targetElement.innerHTML += `
${todayFlag ? '[TODAY] ' : ''}${text}
`; }); } } } function filterSchedule(mode) { document.getElementById('btn-all').classList.toggle('active', mode === 'all'); document.getElementById('btn-today').classList.toggle('active', mode === 'today'); renderDashboard(mode); }
